cmake_minimum_required(VERSION 3.23)
project(MoonlightAppleVideo VERSION 0.1.0 LANGUAGES C CXX)
include(CTest)
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
option(MAV_BUILD_TOOLS "Build macOS offline tools" ON)
option(MAV_SANITIZE "Enable address and undefined behavior sanitizers" OFF)
option(MAV_VT_EXPERIMENTS "Build opt-in VideoToolbox diagnostics (not for production)" OFF)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_OBJCXX_STANDARD 23)
set(CMAKE_OBJCXX_STANDARD_REQUIRED ON)
if(MAV_SANITIZE)
 add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
 add_link_options(-fsanitize=address,undefined)
endif()
add_library(mav-bitstream STATIC src/bitstream.cpp)
target_include_directories(mav-bitstream PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/src>)
set_target_properties(mav-bitstream PROPERTIES POSITION_INDEPENDENT_CODE ON)
set(MAV_CORE src/decoder.cpp src/clock.cpp src/bitstream.cpp)
if(APPLE)
 enable_language(OBJCXX)
 list(APPEND MAV_CORE src/backend_vt.mm src/format.mm src/sample.mm)
else()
 list(APPEND MAV_CORE src/backend_unavailable.cpp)
endif()
add_library(moonlight-apple-video STATIC ${MAV_CORE})
add_library(MoonlightAppleVideo::Decoder ALIAS moonlight-apple-video)
set_target_properties(moonlight-apple-video PROPERTIES EXPORT_NAME Decoder POSITION_INDEPENDENT_CODE ON)
target_include_directories(moonlight-apple-video PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}> PRIVATE src)
find_package(Threads REQUIRED)
target_link_libraries(moonlight-apple-video PUBLIC Threads::Threads)
if(APPLE)
 target_link_libraries(moonlight-apple-video PUBLIC "-framework VideoToolbox" "-framework CoreMedia" "-framework CoreVideo" "-framework CoreFoundation" "-framework IOSurface")
 target_compile_options(moonlight-apple-video PRIVATE $<$<COMPILE_LANGUAGE:OBJCXX>:-fobjc-arc>)
 if(MAV_VT_EXPERIMENTS)
  target_compile_definitions(moonlight-apple-video PRIVATE MAV_VT_EXPERIMENTS=1)
 endif()
endif()
if(BUILD_TESTING)
 add_executable(mav-bitstream-tests tests/bitstream_tests.cpp)
 target_link_libraries(mav-bitstream-tests PRIVATE mav-bitstream)
 add_test(NAME bitstream COMMAND mav-bitstream-tests)
 add_executable(mav-abi-c tests/abi_c.c)
 target_link_libraries(mav-abi-c PRIVATE moonlight-apple-video)
 add_test(NAME abi-c COMMAND mav-abi-c)
 add_executable(mav-adapter-timing-tests tests/adapter_timing_tests.cpp)
 add_test(NAME adapter-timing COMMAND mav-adapter-timing-tests)
 if(APPLE AND MAV_VT_EXPERIMENTS)
  add_executable(mav-vt-controls-tests tests/vt_controls.mm)
  target_include_directories(mav-vt-controls-tests PRIVATE src)
  target_compile_definitions(mav-vt-controls-tests PRIVATE MAV_VT_EXPERIMENTS=1)
  target_compile_options(mav-vt-controls-tests PRIVATE -fobjc-arc)
  target_link_libraries(mav-vt-controls-tests PRIVATE moonlight-apple-video)
  add_test(NAME vt-controls COMMAND mav-vt-controls-tests)
  add_executable(mav-vt-retained-outputs-tests tests/vt_retained_outputs.mm)
  target_compile_definitions(mav-vt-retained-outputs-tests PRIVATE MAV_VT_EXPERIMENTS=1)
  target_compile_options(mav-vt-retained-outputs-tests PRIVATE -fobjc-arc)
  target_link_libraries(mav-vt-retained-outputs-tests PRIVATE "-framework CoreVideo" "-framework CoreFoundation")
  add_test(NAME vt-retained-outputs COMMAND mav-vt-retained-outputs-tests)
 endif()
 if(EXISTS ${CMAKE_CURRENT_SOURCE_DIR}/tests/lifecycle_tests.cpp)
  add_executable(mav-lifecycle-tests tests/lifecycle_tests.cpp tests/backend_fake.cpp src/decoder.cpp src/clock.cpp src/bitstream.cpp)
  target_include_directories(mav-lifecycle-tests PRIVATE include src tests)
  target_link_libraries(mav-lifecycle-tests PRIVATE Threads::Threads)
  add_test(NAME lifecycle COMMAND mav-lifecycle-tests)
 endif()
endif()
if(APPLE AND CMAKE_SYSTEM_NAME STREQUAL "Darwin" AND MAV_BUILD_TOOLS)
 add_executable(mav-reconfiguration tests/reconfiguration.mm)
 target_include_directories(mav-reconfiguration PRIVATE src tools)
 target_compile_options(mav-reconfiguration PRIVATE -fobjc-arc)
 target_link_libraries(mav-reconfiguration PRIVATE moonlight-apple-video "-framework Foundation")
 add_executable(mav-av1-accounting tests/av1_accounting.mm)
 target_include_directories(mav-av1-accounting PRIVATE src)
 target_link_libraries(mav-av1-accounting PRIVATE moonlight-apple-video)
 add_executable(mav-codec-probe tests/apple_probe.mm)
 target_link_libraries(mav-codec-probe PRIVATE moonlight-apple-video "-framework Foundation" "-framework Metal")
 target_compile_options(mav-codec-probe PRIVATE -fobjc-arc)
 foreach(tool fixture replay)
  if(EXISTS ${CMAKE_CURRENT_SOURCE_DIR}/tools/${tool}.mm)
   add_executable(mav-${tool} tools/${tool}.mm)
   target_include_directories(mav-${tool} PRIVATE src)
   target_compile_options(mav-${tool} PRIVATE -fobjc-arc)
   target_link_libraries(mav-${tool} PRIVATE moonlight-apple-video "-framework Foundation" "-framework Metal")
   if(MAV_VT_EXPERIMENTS AND tool STREQUAL "replay")
    target_compile_definitions(mav-${tool} PRIVATE MAV_VT_EXPERIMENTS=1)
   endif()
  endif()
 endforeach()
endif()
install(TARGETS moonlight-apple-video EXPORT MoonlightAppleVideoTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(EXPORT MoonlightAppleVideoTargets NAMESPACE MoonlightAppleVideo:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/MoonlightAppleVideo)
configure_package_config_file(cmake/MoonlightAppleVideoConfig.cmake.in ${CMAKE_CURRENT_BINARY_DIR}/MoonlightAppleVideoConfig.cmake INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/MoonlightAppleVideo)
write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/MoonlightAppleVideoConfigVersion.cmake VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/MoonlightAppleVideoConfig.cmake ${CMAKE_CURRENT_BINARY_DIR}/MoonlightAppleVideoConfigVersion.cmake DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/MoonlightAppleVideo)

# Optional tooling only; FFmpeg never links into the production decoder target.
set(MAV_FFMPEG_ROOT "" CACHE PATH "FFmpeg prefix containing include/ and lib/")
if(APPLE AND CMAKE_SYSTEM_NAME STREQUAL "Darwin" AND MAV_BUILD_TOOLS AND MAV_FFMPEG_ROOT)
 find_path(MAV_AVCODEC_INCLUDE libavcodec/avcodec.h PATHS "${MAV_FFMPEG_ROOT}/include" NO_DEFAULT_PATH REQUIRED)
 find_library(MAV_AVCODEC NAMES avcodec libavcodec.63.dylib libavcodec.62.dylib libavcodec.61.dylib PATHS "${MAV_FFMPEG_ROOT}/lib" NO_DEFAULT_PATH REQUIRED)
 find_library(MAV_AVUTIL NAMES avutil libavutil.61.dylib libavutil.60.dylib libavutil.59.dylib PATHS "${MAV_FFMPEG_ROOT}/lib" NO_DEFAULT_PATH REQUIRED)
 add_executable(mav-ffmpeg-baseline tools/ffmpeg_baseline.mm src/clock.cpp)
 target_include_directories(mav-ffmpeg-baseline PRIVATE include "${MAV_AVCODEC_INCLUDE}")
 target_compile_options(mav-ffmpeg-baseline PRIVATE $<$<COMPILE_LANGUAGE:OBJCXX>:-fobjc-arc>)
 target_link_libraries(mav-ffmpeg-baseline PRIVATE "${MAV_AVCODEC}" "${MAV_AVUTIL}" "-framework Foundation" "-framework VideoToolbox" "-framework CoreVideo" "-framework CoreMedia")
 add_executable(mav-reference-decode tools/reference_decode.mm)
 target_include_directories(mav-reference-decode PRIVATE "${MAV_AVCODEC_INCLUDE}")
 target_compile_options(mav-reference-decode PRIVATE -fobjc-arc)
 target_link_libraries(mav-reference-decode PRIVATE "${MAV_AVCODEC}" "${MAV_AVUTIL}" "-framework Foundation")
endif()
